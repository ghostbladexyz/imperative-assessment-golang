// getJSON GETs url, checks the status code, and decodes the body into v. It
// returns an error (never panics) on a transport failure, a non-200 status,
// or a body that doesn't unmarshal into v.
func getJSON(url string, v any) error {
	resp, err := http.Get(url)
	if err != nil {
		return err
	}
	defer resp.Body.Close()
	if resp.StatusCode != http.StatusOK {
		return fmt.Errorf("GET %s: status %d", url, resp.StatusCode)
	}
	b, err := io.ReadAll(resp.Body)
	if err != nil {
		return err
	}
	return json.Unmarshal(b, v)
}
