package main

import (
	"fmt"
	"os"
	"strings"
)

// Colony is the parsed ant farm. Adj maps each room to its neighbours.
type Colony struct {
	Start, End string
	Adj        map[string][]string
}

func parseColony(path string) (*Colony, error) {
	data, err := os.ReadFile(path)
	if err != nil {
		return nil, err
	}
	c := &Colony{Adj: map[string][]string{}}
	pending := ""
	for _, raw := range strings.Split(string(data), "\n") {
		line := strings.TrimSpace(raw)
		if line == "" {
			continue
		}
		switch line {
		case "##start":
			pending = "start"
			continue
		case "##end":
			pending = "end"
			continue
		}
		if strings.HasPrefix(line, "#") { // a comment
			continue
		}
		if strings.Contains(line, "-") && !strings.Contains(line, " ") { // a tunnel a-b
			p := strings.SplitN(line, "-", 2)
			c.Adj[p[0]] = append(c.Adj[p[0]], p[1])
			c.Adj[p[1]] = append(c.Adj[p[1]], p[0])
			continue
		}
		if f := strings.Fields(line); len(f) == 3 { // a room "name x y"
			if _, ok := c.Adj[f[0]]; !ok {
				c.Adj[f[0]] = nil
			}
			switch pending {
			case "start":
				c.Start = f[0]
			case "end":
				c.End = f[0]
			}
			pending = ""
		}
		// the lone ant-count line is ignored: a single path does not need it
	}
	return c, nil
}
